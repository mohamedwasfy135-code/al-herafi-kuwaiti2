  prisma: PrismaClient | undefined
}

// دالة لضمان وجود pgbouncer=true في الرابط
function getDatabaseUrl() {
  let url = process.env.DATABASE_URL || ''
  if (!url) return url
  // إذا كان الرابط لا يحتوي على pgbouncer=true، نضيفه
  if (!url.includes('pgbouncer=true')) {
    const separator = url.includes('?') ? '&' : '?'
    url = `${url}${separator}pgbouncer=true`
  }
  console.log('[DB] Effective URL prefix:', url.substring(0, 80) + '...')
  return url
}

export const db = globalForPrisma.prisma ?? new PrismaClient({
  log: ['error'],
  datasourceUrl: getDatabaseUrl(),
})

if (process.env.NODE_ENV !== 'production') globalForPrisma.prisma = db

import { db } from '@/lib/db'

/**
 * Resolves a businessId that might be a user ID (ownerId) to the actual Business.id
 * This handles backward compatibility where some clients may send userId instead of businessId
 */
export async function resolveBusinessId(id: string): Promise<string | null> {
  if (!id) return null

  // First try as a direct business ID
  const directBusiness = await db.business.findUnique({ where: { id }, select: { id: true } })
  if (directBusiness) return directBusiness.id

  // Then try as owner ID
  const byOwner = await db.business.findFirst({ where: { ownerId: id }, select: { id: true } })
  if (byOwner) return byOwner.id

  return null
}

export type AuditAction = 'CREATE' | 'UPDATE' | 'DELETE' | 'PAY' | 'MERGE' | 'EXPORT' | 'IMPORT'

export async function logAudit({
  businessId,
  userId,
  action,
  entity,
  entityId,
  changes,
  ipAddress,
}: {
  businessId: string
  userId?: string
  action: AuditAction
  entity: string
  entityId?: number
  changes?: { before?: unknown; after?: unknown }
  ipAddress?: string
}) {
  try {
    await db.auditLog.create({
      data: {
        businessId,
        userId,
        action,
        entity,
        entityId,
        changes: changes ? JSON.parse(JSON.stringify(changes)) : undefined,
        ipAddress,
      },
    })
  } catch (error) {
    console.error('[AUDIT] Failed to log:', error)
  }
}

import { SignJWT, jwtVerify } from 'jose'
import { cookies } from 'next/headers'

const secretKey = process.env.JWT_SECRET || 'sana3i-kw-super-secret-key-2024'
const key = new TextEncoder().encode(secretKey)

type SessionData = {
  userId: string
  role: string
  name: string
}

// إنشاء JWT token وإرجاعه (بدون تعيين كوكي)
export async function createSessionToken(data: SessionData): Promise<string> {
  const token = await new SignJWT(data)
    .setProtectedHeader({ alg: 'HS256' })
    .setIssuedAt()
    .setExpirationTime('7d')
    .sign(key)

  return token
}

// إعدادات الكوكي
const cookieOptions = {
  httpOnly: true,
  secure: process.env.NODE_ENV === 'production',
  sameSite: 'lax' as const,
  maxAge: 60 * 60 * 24 * 7, // 7 أيام
  path: '/',
}

// تعيين كوكي الجلسة على استجابة NextResponse
export function setSessionCookie(response: Response, token: string) {
  response.headers.append('Set-Cookie', `sana3i_session=${token}; HttpOnly; ${process.env.NODE_ENV === 'production' ? 'Secure; ' : ''}SameSite=Lax; Max-Age=${cookieOptions.maxAge}; Path=/`)
}

// حذف كوكي الجلسة من استجابة NextResponse
export function clearSessionCookie(response: Response) {
  response.headers.append('Set-Cookie', `sana3i_session=; HttpOnly; ${process.env.NODE_ENV === 'production' ? 'Secure; ' : ''}SameSite=Lax; Max-Age=0; Path=/`)
}

// قراءة الجلسة من الكوكي (للمiddleware والصفحات)
export async function getSession(): Promise<SessionData | null> {
  const cookieStore = await cookies()
  const token = cookieStore.get('sana3i_session')?.value

  if (!token) return null

  try {
    const { payload } = await jwtVerify(token, key)
    return payload as SessionData
  } catch {
    return null
  }
}

// الحصول على الجلسة من Request (للمiddleware)
export async function getSessionFromRequest(request: Request): Promise<SessionData | null> {
  const token = request.headers.get('cookie')
    ?.split('; ')
    .find(c => c.startsWith('sana3i_session='))
    ?.split('=')[1]

  if (!token) return null

  try {
    const { payload } = await jwtVerify(token, key)
    return payload as SessionData
  } catch {
    return null
  }
}

// دالة قديمة للتوافق - لكنها الآن تعمل بشكل مختلف
export async function createSession(data: SessionData) {
  const token = await createSessionToken(data)
  const cookieStore = await cookies()
  cookieStore.set('sana3i_session', token, cookieOptions)
}

import { NextRequest, NextResponse } from 'next/server'
import { Pool } from 'pg'

const pool = new Pool({ connectionString: process.env.DATABASE_URL, ssl: { rejectUnauthorized: false } })

export async function POST(request: NextRequest) {
  try {
    const { requestId, craftsmanId } = await request.json()
    if (!requestId || !craftsmanId) {
      return NextResponse.json({ error: 'requestId and craftsmanId required' }, { status: 400 })
    }

    // حذف الإسنادات السابقة لهذا الطلب
    await pool.query(`DELETE FROM request_assignments WHERE request_id = $1`, [requestId])

    // إضافة إسناد جديد بحالة pending
    const expiresAt = new Date(Date.now() + 30 * 60 * 1000)
    await pool.query(
      `INSERT INTO request_assignments (request_id, craftsman_id, status, expires_at, created_at, updated_at)
       VALUES ($1, $2, 'pending', $3, NOW(), NOW())`,
      [requestId, craftsmanId, expiresAt]
    )

    // تحديث الطلب
    await pool.query(
      `UPDATE requests SET craftsman_id = $1, status = 'bidding', updated_at = NOW() WHERE id = $2`,
      [craftsmanId, requestId]
    )

    // إشعار (بدون related_id)
    await pool.query(
      `INSERT INTO notifications (user_id, title, body, type)
       VALUES ($1, '📢 طلب تسعير جديد', 'تم إسناد طلب رقم $2 إليك للتسعير', 'bid_request')`,
      [craftsmanId, requestId]
    )

    return NextResponse.json({ success: true })
  } catch (error: any) {
    console.error('Force assign error:', error)
    return NextResponse.json({ error: error.message }, { status: 500 })
  }
}

import { NextResponse } from 'next/server'
import { PrismaClient } from '@prisma/client'

const prisma = new PrismaClient()

export async function GET(request: Request) {
  try {
    const { searchParams } = new URL(request.url)
    const categoriesParam = searchParams.get('categories') // مثال: "سباكة,كهرباء"
    let categories: string[] = []

    if (categoriesParam) {
      categories = categoriesParam.split(',').map(c => c.trim())
    }

    const whereClause: any = {
      craftsmanId: null,
      status: { in: ['pending', 'bidding'] }
    }

    if (categories.length > 0) {
      whereClause.serviceType = { in: categories }
    }

    const openRequests = await prisma.request.findMany({
      where: whereClause,
      orderBy: { createdAt: 'desc' }
    })

    return NextResponse.json({ requests: openRequests })
  } catch (error: any) {
    console.error('Open requests error:', error)
    return NextResponse.json({ error: error.message }, { status: 500 })
  }
}

    }
  }

  const updateUserStatus = async (userId: string, field: string, value: string) => {
    try {
      await fetch('/api/admin/users', { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ userId, field, value }) })
      await loadAll()
      setMsg(`✅ تم التحديث`)
      setTimeout(() => setMsg(''), 3000)
    } catch (err) { console.error(err) }
  }

  const updateRequestStatus = async (requestId: string, status: string) => {
    try {
      await fetch('/api/admin/requests', { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ requestId, status }) })
      await loadAll()
      setMsg(`✅ تم تحديث حالة الطلب`)
      setTimeout(() => setMsg(''), 3000)
    } catch (err) { console.error(err) }
  }

  const assignRequest = async () => {
    if (!assignModal || !selectedCraftsman) return
    try {
      await fetch('/api/admin/requests', { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ requestId: assignModal.id, craftsmanId: selectedCraftsman, status: 'accepted' }) })
      setAssignModal(null)
      setSelectedCraftsman('')
      await loadAll()
      setMsg('✅ تم إسناد الطلب')
      setTimeout(() => setMsg(''), 3000)
    } catch (err) { console.error(err) }
  }

  const reassignRequest = async () => {
    if (!reassignModal || !selectedCraftsman) {
      setMsg("❌ يرجى اختيار حرفي جديد");
      setTimeout(() => setMsg(""), 3000);
      return;
    }
    if (!reassignModal || !selectedCraftsman) return
    try {
      const res = await fetch('/api/admin/reassign-request', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ requestId: reassignModal.id, newCraftsmanId: selectedCraftsman, oldCraftsmanId: reassignModal.craftsman_id })
      })
      if (res.ok) {
        setMsg('✅ تم إعادة الإسناد وإرسال الإشعارات')
        setReassignModal(null)
        setSelectedCraftsman('')
        await loadAll()
      } else {
        setMsg('❌ فشل إعادة الإسناد')
      }
      setTimeout(() => setMsg(''), 3000)
    } catch (err) { console.error(err) }
  }

  const updatePayout = async (payoutId: string, status: string) => {
    try {
      await fetch('/api/admin/payouts', { method: 'PUT', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ payoutId, status }) })
      await loadAll()
      setMsg(status === 'approved' ? '✅ تمت الموافقة' : '❌ تم الرفض')
      setTimeout(() => setMsg(''), 3000)
    } catch (err) { console.error(err) }
  }

  const handleChangeRequest = async (requestId: number, action: 'approve' | 'reject') => {
    try {
      const res = await fetch('/api/admin/change-requests', {
        method: 'PUT',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ requestId, action })
      })
      if (res.ok) {
        setMsg(action === 'approve' ? '✅ تمت الموافقة على تغيير الحرفة' : '❌ تم رفض طلب تغيير الحرفة')
        await loadAll()
      } else {
        setMsg('❌ حدث خطأ')
      }
      setTimeout(() => setMsg(''), 3000)
    } catch (err) { console.error(err) }
  }

  const handleDocumentAction = async (documentId: number, action: 'approve' | 'reject') => {
    try {
      const res = await fetch('/api/admin/documents', {
        method: 'PUT',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ documentId, action, adminId: admin?.id })
      })
      if (res.ok) {
        setMsg(action === 'approve' ? '✅ تم اعتماد المستندات' : '❌ تم رفض المستندات')
        await loadAll()
      } else {
        setMsg('❌ حدث خطأ')
      }
      setTimeout(() => setMsg(''), 3000)
    } catch (err) { console.error(err) }
  }

  const handleRefundAction = async (refundId: number, action: 'approve' | 'reject', adminNotes?: string) => {
    try {
      const res = await fetch('/api/admin/refund-requests', {
        method: 'PUT',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ refundId, action, adminNotes })
      })
      if (res.ok) {
        setMsg(action === 'approve' ? '✅ تمت الموافقة على الاسترداد' : '❌ تم رفض الاسترداد')
        await loadAll()
      } else {
        setMsg('❌ حدث خطأ')
      }
      setTimeout(() => setMsg(''), 3000)
    } catch (err) { console.error(err) }
  }

  const handleLogout = async () => {
    await fetch('/api/auth/logout', { method: 'POST' })
    localStorage.removeItem('sana3i_user')
    router.push('/login')
  }

  if (loading) return <div className="p-8 text-center">جار التحميل...</div>
  if (!admin) return <div className="p-8 text-center">يرجى تسجيل الدخول...</div>

  return (
    <div dir="rtl" className="min-h-screen flex bg-gray-100">
      <aside className="w-56 bg-gray-900 text-white p-4">
        <h2 className="text-lg font-bold mb-6">لوحة الإدارة</h2>
        <nav className="space-y-2">
          {[
            { key: 'overview', label: '📊 نظرة عامة' },
            { key: 'users', label: '👥 المستخدمون' },
            { key: 'requests', label: '📋 الطلبات' },
            { key: 'earnings', label: '💰 الأرباح' },
            { key: 'reports', label: '📈 تقارير ذكية' },
            { key: 'changeRequests', label: '🔄 طلبات تغيير الحرفة' },
            { key: 'documents', label: '📄 اعتماد المستندات' },
            { key: 'refunds', label: '💰 طلبات الاسترداد' },
            { key: 'interventions', label: '🛠️ تدخلات الأدمن' },
          ].map(item => (
            <button key={item.key} onClick={() => setTab(item.key)}
              className={`w-full text-right px-3 py-2 rounded-lg text-sm ${tab === item.key ? 'bg-blue-600' : 'hover:bg-gray-800'}`}>
              {item.label}
            </button>
          ))}
        </nav>
        <button onClick={handleLogout} className="mt-6 text-sm text-red-400 hover:underline">تسجيل الخروج</button>
      </aside>

      <main className="flex-1 p-6">
        <h1 className="text-2xl font-bold mb-6">مرحباً، {admin.name}</h1>
        {msg && <div className="bg-green-100 text-green-700 p-3 rounded mb-4">{msg}</div>}

        {tab === 'overview' && stats && (
          <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">
            <div className="bg-white rounded-xl shadow p-4"><p className="text-gray-500">المستخدمون</p><p className="text-2xl font-bold">{stats.users}</p></div>
            <div className="bg-white rounded-xl shadow p-4"><p className="text-gray-500">الطلبات</p><p className="text-2xl font-bold">{stats.requests}</p></div>
            <div className="bg-white rounded-xl shadow p-4"><p className="text-gray-500">الإيرادات</p><p className="text-2xl font-bold">{stats.revenue} د.ك</p></div>
            <div className="bg-white rounded-xl shadow p-4"><p className="text-gray-500">حرفيون بانتظار الاعتماد</p><p className="text-2xl font-bold text-red-500">{stats.pendingCraftsmen}</p></div>
          </div>
        )}

        {tab === 'users' && (
          <div className="bg-white rounded-xl shadow overflow-x-auto">
            <table className="w-full text-sm">
              <thead className="bg-gray-50"><tr><th className="p-3 text-right">الاسم</th><th className="p-3 text-right">الدور</th><th className="p-3 text-right">حالة الاستقبال</th><th className="p-3 text-right">حالة الاعتماد</th><th className="p-3 text-right">إجراء</th></tr></thead>
              <tbody>
                {users.map((u: any) => (
                  <tr key={u.id} className="border-t">
                    <td className="p-3">{u.name}</td>
                    <td className="p-3">{u.role}</td>
                    <td className="p-3">
                      {u.role === 'craftsman' ? (
                        u.is_available ? '🟢 متاح' : (u.unavailable_until ? `🔴 غير متاح حتى ${new Date(u.unavailable_until).toLocaleTimeString('ar')}` : '🔴 غير متاح')
                      ) : '-'}
                    </td>
                    <td className="p-3">{u.verification_status || '-'}</td>
                    <td className="p-3">
                      {u.role === 'craftsman' && u.verification_status !== 'approved' && (
                        <button onClick={() => updateUserStatus(u.id, 'verification_status', 'approved')} className="text-xs bg-green-600 text-white px-2 py-1 rounded">اعتماد</button>
                      )}
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        )}

        {tab === 'requests' && (
          <div className="bg-white rounded-xl shadow overflow-x-auto">
            <table className="w-full text-sm">
              <thead className="bg-gray-50"><tr><th className="p-3 text-right">الخدمة</th><th className="p-3 text-right">العميل</th><th className="p-3 text-right">الحرفي</th><th className="p-3 text-right">الحالة</th><th className="p-3 text-right">إجراءات</th></tr></thead>
              <tbody>
                {requests.map((r: any) => (
                  <tr key={r.id} className="border-t">
                    <td className="p-3">{r.service_type || 'طلب'}</td>
                    <td className="p-3">{r.client?.name || '—'}</td>
                    <td className="p-3">{r.craftsman?.name || '—'}</td>
                    <td className="p-3">{r.status}</td>
                    <td className="p-3 flex gap-2">
                      {!r.craftsman && (
                        <button onClick={() => { setAssignModal(r); setSelectedCraftsman('') }} className="text-xs bg-purple-600 text-white px-2 py-1 rounded">إسناد يدوي</button>
                      )}
                      {r.craftsman && (r.status === 'pending' || r.status === 'accepted') && (
                        <button onClick={() => { setReassignModal(r); setSelectedCraftsman('') }} className="text-xs bg-orange-600 text-white px-2 py-1 rounded">إعادة إسناد</button>
                      )}
                      <select onChange={e => updateRequestStatus(r.id, e.target.value)} className="text-xs border rounded px-2 py-1" defaultValue="">
                        <option disabled>تغيير الحالة</option>
                        <option value="pending">قيد الانتظار</option><option value="accepted">مقبول</option><option value="in_progress">قيد التنفيذ</option><option value="done">مكتمل</option><option value="rejected">مرفوض</option>
                      </select>
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        )}

        {tab === 'earnings' && (
          <div>
            <div className="bg-white rounded-xl shadow p-4 mb-6">
              <h2 className="text-xl font-bold mb-4">ملخص الأرباح</h2>
              <div className="grid grid-cols-3 gap-4">
                <div><p className="text-gray-500">إجمالي الأرباح</p><p className="text-2xl font-bold text-green-600">{earnings.reduce((sum, e) => sum + (parseFloat(e.amount) || 0), 0).toFixed(2)} د.ك</p></div>
                <div><p className="text-gray-500">عدد العمليات</p><p className="text-2xl font-bold">{earnings.length}</p></div>
                <div><p className="text-gray-500">طلبات دفعات معلقة</p><p className="text-2xl font-bold text-yellow-600">{payouts.filter(p => p.status === 'pending').length}</p></div>
              </div>
            </div>
            <div className="bg-white rounded-xl shadow overflow-x-auto">
              <table className="w-full text-sm">
                <thead className="bg-gray-50"><tr><th className="p-3 text-right">الحرفي</th><th className="p-3 text-right">المبلغ</th><th className="p-3 text-right">الحالة</th><th className="p-3 text-right">إجراء</th></tr></thead>
                <tbody>
                  {payouts.map((p: any) => (
                    <tr key={p.id} className="border-t">
                      <td className="p-3">{p.craftsman_name}</td><td className="p-3 font-bold">{p.amount} د.ك</td><td className="p-3">{p.status}</td>
                      <td className="p-3 flex gap-2">
                        {p.status === 'pending' && (<><button onClick={() => updatePayout(p.id, 'approved')} className="text-xs bg-green-600 text-white px-2 py-1 rounded">موافقة</button><button onClick={() => updatePayout(p.id, 'rejected')} className="text-xs bg-red-500 text-white px-2 py-1 rounded">رفض</button></>)}
                      </td>
                    </tr>
                  ))}
                </tbody>
              </table>
            </div>
          </div>
        )}

        {tab === 'reports' && (
          <div className="bg-white rounded-xl shadow p-4">
            <h2 className="text-xl font-bold mb-4">التقارير الذكية</h2>
            <p className="text-gray-500">قيد التطوير (ستظهر قريباً نقاط القوة والضعف وأداء الفئات)</p>
          </div>
        )}

        {tab === 'changeRequests' && (
          <div className="bg-white rounded-xl shadow overflow-x-auto">
            <h2 className="text-xl font-bold p-4 border-b">طلبات تغيير الحرفة</h2>
            {changeRequests.length === 0 ? (
              <div className="p-8 text-center text-gray-400">لا توجد طلبات معلقة</div>
            ) : (
              <table className="w-full text-sm">
                <thead className="bg-gray-50"><tr><th className="p-3 text-right">الحرفي</th><th className="p-3 text-right">الهاتف</th><th className="p-3 text-right">الحرفة الحالية</th><th className="p-3 text-right">الحرفة الجديدة</th><th className="p-3 text-right">السبب</th><th className="p-3 text-right">إجراء</th></tr></thead>
                <tbody>
                  {changeRequests.map((req: any) => (
                    <tr key={req.id} className="border-t">
                      <td className="p-3">{req.craftsman_name}</td>
                      <td className="p-3">{req.phone}</td>
                      <td className="p-3">{req.old_category_name || '—'}</td>
                      <td className="p-3">{req.new_category_name}</td>
                      <td className="p-3 max-w-xs truncate">{req.reason}</td>
                      <td className="p-3 flex gap-2">
                        <button onClick={() => handleChangeRequest(req.id, 'approve')} className="text-xs bg-green-600 text-white px-2 py-1 rounded">موافقة</button>
                        <button onClick={() => handleChangeRequest(req.id, 'reject')} className="text-xs bg-red-500 text-white px-2 py-1 rounded">رفض</button>
                      </td>
                    </tr>
                  ))}
                </tbody>
              </table>
            )}
          </div>
        )}

        {tab === 'documents' && (
          <div className="bg-white rounded-xl shadow overflow-x-auto">
            <h2 className="text-xl font-bold p-4 border-b">مستندات الحرفيين بانتظار الاعتماد</h2>
            {pendingDocs.length === 0 ? (
              <div className="p-8 text-center text-gray-400">لا توجد مستندات معلقة</div>
            ) : (
              <table className="w-full text-sm">
                <thead className="bg-gray-50"><tr><th className="p-3 text-right">الحرفي</th><th className="p-3 text-right">الهاتف</th><th className="p-3 text-right">البطاقة المدنية</th><th className="p-3 text-right">الحساب البنكي</th><th className="p-3 text-right">إجراء</th></tr></thead>
                <tbody>
                  {pendingDocs.map((doc: any) => (
                    <tr key={doc.id} className="border-t">
                      <td className="p-3">{doc.craftsman_name}</td>
                      <td className="p-3">{doc.phone}</td>
                      <td className="p-3">
                        {doc.civil_id_url ? <a href={doc.civil_id_url} target="_blank" className="text-blue-500 underline">عرض</a> : '—'}
                      </td>
                      <td className="p-3">
                        {doc.bank_account_photo_url ? <a href={doc.bank_account_photo_url} target="_blank" className="text-blue-500 underline">عرض</a> : '—'}
                        {doc.bank_name && <div className="text-xs text-gray-500">{doc.bank_name} - {doc.bank_iban}</div>}
                      </td>
                      <td className="p-3 flex gap-2">
                        <button onClick={() => handleDocumentAction(doc.id, 'approve')} className="text-xs bg-green-600 text-white px-2 py-1 rounded">اعتماد</button>
                        <button onClick={() => handleDocumentAction(doc.id, 'reject')} className="text-xs bg-red-500 text-white px-2 py-1 rounded">رفض</button>
                      </td>
                    </tr>
                  ))}
                </tbody>
              </table>
            )}
          </div>
        )}

        {tab === 'refunds' && (
          <div className="bg-white rounded-xl shadow overflow-x-auto">
            <h2 className="text-xl font-bold p-4 border-b">طلبات الاسترداد</h2>
            {pendingRefunds.length === 0 ? (
              <div className="p-8 text-center text-gray-400">لا توجد طلبات استرداد معلقة</div>
            ) : (
              <table className="w-full text-sm">
                <thead className="bg-gray-50"><tr><th className="p-3 text-right">العميل</th><th className="p-3 text-right">الطلب</th><th className="p-3 text-right">المبلغ المسترد</th><th className="p-3 text-right">السبب</th><th className="p-3 text-right">إجراء</th></tr></thead>
                <tbody>
                  {pendingRefunds.map((ref: any) => (
                    <tr key={ref.id} className="border-t">
                      <td className="p-3">{ref.client_name}</td>
                      <td className="p-3">{ref.service_type || ref.request_id}</td>
                      <td className="p-3 font-bold text-red-600">{ref.amount} د.ك</td>
                      <td className="p-3 max-w-xs truncate">{ref.reason || '—'}</td>
                      <td className="p-3 flex gap-2">
                        <button onClick={() => handleRefundAction(ref.id, 'approve')} className="text-xs bg-green-600 text-white px-2 py-1 rounded">موافقة</button>
                        <button onClick={() => handleRefundAction(ref.id, 'reject')} className="text-xs bg-red-500 text-white px-2 py-1 rounded">رفض</button>
                      </td>
                    </tr>
                  ))}
                </tbody>
              </table>
            )}
          </div>
        )}

        {tab === 'interventions' && (
          <div className="bg-white rounded-xl shadow overflow-x-auto">
            <h2 className="text-xl font-bold p-4 border-b">طلبات محولة للأدمن</h2>
            {interventionRequests.length === 0 ? (
              <div className="p-8 text-center text-gray-400">لا توجد طلبات محولة</div>
            ) : (
              <table className="w-full text-sm">
                <thead className="bg-gray-50"><tr><th className="p-3 text-right">العميل</th><th className="p-3 text-right">الخدمة</th><th className="p-3 text-right">السبب</th><th className="p-3 text-right">إجراء</th></tr></thead>
                <tbody>
                  {interventionRequests.map((req: any) => (
                    <tr key={req.id} className="border-t">
                      <td className="p-3">{req.client_name}</td>
                      <td className="p-3">{req.service_type}</td>
                      <td className="p-3">{req.intervention_reason || 'رفض العميل كل العروض'}</td>
                      <td className="p-3 flex gap-2">
                        <button onClick={() => { setAssignModal(req); setSelectedCraftsman('') }} className="text-xs bg-purple-600 text-white px-2 py-1 rounded">إسناد يدوي</button>
                      </td>
                    </tr>
                  ))}
                </tbody>
              </table>
            )}
          </div>
        )}
      </main>

      {assignModal && (
        <div className="fixed inset-0 bg-black bg-opacity-50 flex items-center justify-center z-50">
          <div className="bg-white rounded-2xl shadow-xl p-6 max-w-md w-full mx-4">
            <h3 className="text-lg font-bold mb-4">إسناد يدوي</h3>
            <p className="text-sm text-gray-500 mb-4">الطلب: {assignModal.service_type || 'طلب خدمة'}</p>
            <select value={selectedCraftsman} onChange={e => setSelectedCraftsman(e.target.value)} className="w-full border rounded-lg px-3 py-2 mb-4">
              <option value="">اختر حرفي</option>
              {craftsmen.map((c: any) => (<option key={c.id} value={c.id}>{c.name} - {c.phone}</option>))}
            </select>
            <div className="flex gap-2">
              <button onClick={assignRequest} disabled={!selectedCraftsman} className="flex-1 bg-purple-600 text-white py-2 rounded-lg disabled:opacity-50">تأكيد</button>
              <button onClick={() => setAssignModal(null)} className="px-4 py-2 bg-gray-100 rounded-lg">إلغاء</button>
            </div>
          </div>
        </div>
      )}

      {reassignModal && (
        <div className="fixed inset-0 bg-black bg-opacity-50 flex items-center justify-center z-50">
          <div className="bg-white rounded-2xl shadow-xl p-6 max-w-md w-full mx-4">
            <h3 className="text-lg font-bold mb-4">إعادة إسناد طلب</h3>
            <p className="text-sm text-gray-500 mb-4">الطلب: {reassignModal.service_type || 'طلب خدمة'} (حالياً مع {reassignModal.craftsman?.name || 'حرفي'})</p>
            <select value={selectedCraftsman} onChange={e => setSelectedCraftsman(e.target.value)} className="w-full border rounded-lg px-3 py-2 mb-4">
              <option value="">اختر حرفي جديد</option>
              {craftsmen.filter(c => c.id !== reassignModal.craftsman_id).map((c: any) => (<option key={c.id} value={c.id}>{c.name} - {c.phone}</option>))}
            </select>
            <div className="flex gap-2">
              <button onClick={reassignRequest} disabled={!selectedCraftsman} className="flex-1 bg-orange-600 text-white py-2 rounded-lg disabled:opacity-50">تأكيد إعادة الإسناد</button>
              <button onClick={() => setReassignModal(null)} className="px-4 py-2 bg-gray-100 rounded-lg">إلغاء</button>
            </div>
          </div>
        </div>
      )}
    </div>
  )
}
